   136→def find_maps_dir(maps_dir=None):
   137→    """Найти папку с PNG картами."""
   138→    candidates = []
   139→    if maps_dir:
   140→        candidates.append(maps_dir)
   141→    candidates.append('.')
   142→    candidates.append(os.path.dirname(os.path.abspath(__file__)) or '.')
   143→
   144→    # Стандартные пути Minecraft
   145→    if sys.platform == "win32":
   146→        appdata = os.environ.get("APPDATA", "")
   147→        if appdata:
   148→            candidates.append(os.path.join(appdata, ".tlauncher", "minecraft"))
   149→            candidates.append(os.path.join(appdata, ".minecraft"))
   150→        localappdata = os.environ.get("LOCALAPPDATA", "")
   151→        if localappdata:
   152→            candidates.append(os.path.join(localappdata, ".minecraft"))
   153→    else:
   154→        home = os.path.expanduser("~")
   155→        candidates.append(home)
   156→
   157→    for d in candidates:
   158→        if os.path.isdir(d) and os.path.exists(os.path.join(d, DEFAULT_MAP_FILES['height'])):
   159→            return d
   160→    return None

   163→def load_maps(maps_dir):
   164→    """Загрузить все 7 PNG карт. Возвращает словарь numpy массивов."""
   165→    maps = {}
   166→    print(f"[*] Загрузка карт из: {maps_dir}")

   169→    # Сначала покажем все файлы в директории
   170→    try:
   171→        all_files = sorted(os.listdir(maps_dir))
   172→        png_files = [f for f in all_files if f.lower().endswith('.png')]
   173→        print(f"    PNG файлы в директории ({len(png_files)}):")
   174→        for f in png_files:
   175→            size = os.path.getsize(os.path.join(maps_dir, f))

   178→        print(f"    [!] Не удалось перечислить файлы: {e}")
   179→
   180→    for key, fname in DEFAULT_MAP_FILES.items():
   181→        path = os.path.join(maps_dir, fname)
   182→        if not os.path.exists(path):
   183→            print(f"    [!] НЕ НАЙДЕН: {fname}")

   185→        try:
   186→            if key == 'temperature':
   187→                img = Image.open(path).convert('RGB')
   188→                maps[key] = np.array(img)
   189→            elif key in ('rivers', 'lakes'):
   190→                # rivers/lakes — RGBA, используем alpha channel
   191→                img = Image.open(path).convert('RGBA')
   192→                maps[key] = np.array(img)
   193→            elif key == 'sea_ocean':
   194→                # v4: sea_ocean — LA или RGBA маска (alpha>0 = океан)
   195→                img = Image.open(path).convert('RGBA')
   196→                maps[key] = np.array(img)

   202→            print(f"    [!] Ошибка загрузки {fname}: {e}")

   208→def detect_climate(rgb):
   209→    """Найти ближайшую климатическую зону к RGB цвету."""
   210→    best = None
   211→    best_dist = float('inf')
   212→    for name, color in CLIMATE_ZONES.items():
   213→        d = np.sqrt(np.sum((np.array(rgb).astype(int) - color) ** 2))
   214→        if d < best_dist:
   215→            best_dist = d
   216→            best = name
   217→    return best, best_dist

   220→def generate_biome_map(maps):
   221→    """Сгенерировать карту биомов (RGB numpy array).
   222→
   223→    v4 (July 2026):
   224→      - Океан определяется МАСКОЙ map_sea-ocean.png (alpha>0 = океан).
   225→      - Глубокий океан: ocean mask И height gray < 42 (Y < 25).
   226→      - Биомы суши: nearest-point matching по параметрам T/H/C/E/W
   227→        (совпадает с Java BiomeParameters.java).
   228→    """
   229→    print("[*] Генерация карты биомов (v4 — mask-driven ocean + multi-noise biomes)...")
   230→
   231→    height_img = maps['height']
   232→    humidity_img = maps['humidity']
   233→    erosion_img = maps['erosion']
   234→    temp_img = maps['temperature']
   235→    lakes_data = maps['lakes']
   236→    rivers_data = maps['rivers']
   237→    sea_ocean_data = maps['sea_ocean']
   238→
   239→    H, W = height_img.shape
   240→    biome_map = np.zeros((H, W, 3), dtype=np.uint8)
   241→
   242→    # ─── Маски воды (v4: ocean from mask) ───
   243→    # Ocean mask: alpha > 0 AND luminance < 128 (LA format)
   244→    ocean_mask = (sea_ocean_data[:, :, 3] > 0) & (sea_ocean_data[:, :, 0] < WATER_RED_THRESHOLD)
   245→    # Deep ocean: ocean AND height < deep_ocean_gray (Y < 25)
   246→    deep_ocean_gray = int(round((DEEP_OCEAN_Y - MIN_OCEAN) * SEA_LEVEL_GRAY / (SEA_LEVEL - MIN_OCEAN)))
   247→    deep_ocean_mask = ocean_mask & (height_img < deep_ocean_gray)
   248→    shallow_ocean_mask = ocean_mask & ~deep_ocean_mask
   249→
   250→    # rivers/lakes: alpha > 0 AND red < 128
   251→    lakes_mask = (lakes_data[:, :, 3] > 0) & (lakes_data[:, :, 0] < WATER_RED_THRESHOLD)
   252→    rivers_mask = (rivers_data[:, :, 3] > 0) & (rivers_data[:, :, 0] < WATER_RED_THRESHOLD)
   253→
   254→    water_mask = ocean_mask | lakes_mask | rivers_mask
   255→    land_mask = ~water_mask
   256→
   257→    # Заполняем воду
   258→    biome_map[deep_ocean_mask] = BIOME_COLORS['deep_ocean']
   259→    biome_map[shallow_ocean_mask] = BIOME_COLORS['ocean']
   260→    biome_map[lakes_mask] = BIOME_COLORS['lake']
   261→    biome_map[rivers_mask] = BIOME_COLORS['river']
   262→
   263→    # ─── Параметры суши для biome matching ───
   264→    # Temperature normalized: ClimateZone.temp (0.05..1.0) → (-1.0, 1.0)
   265→    def get_climate(temp_rgb):
   266→        best = None
   267→        best_dist = float('inf')
   268→        for name, color in CLIMATE_ZONES.items():
   269→            d = np.sqrt(np.sum((np.array(temp_rgb).astype(int) - color) ** 2))
   270→            if d < best_dist:
   271→                best_dist = d
   272→                best = name
   273→        return best
   274→
   275→    temp_norm_grid = np.zeros((H, W), dtype=np.float32)
   276→    climate_grid = np.empty((H, W), dtype='U20')
   277→    for z in range(H):
   278→        for x in range(W):
   279→            climate = get_climate(temp_img[z, x])
   280→            climate_grid[z, x] = climate
   281→            temp_norm_grid[z, x] = (CLIMATE_TEMPS[climate] - 0.5) * 2.0
   282→
   283→    # Humidity normalized: 0..255 → (-1.0, 1.0)
   284→    hum_norm_grid = (humidity_img.astype(np.float32) / 255.0) * 2.0 - 1.0
   285→
   286→    # Erosion normalized: 0..255 → (-1.0, 1.0)
   287→    ero_norm_grid = (erosion_img.astype(np.float32) / 255.0) * 2.0 - 1.0
   288→
   289→    # Continentalness: from ocean mask proximity
   290→    # Ocean pixel → C ≈ -1.0, Land far from coast → C ≈ 0.6
   291→    from scipy.ndimage import gaussian_filter
   292→    ocean_float = ocean_mask.astype(np.float32)
   293→    ocean_smooth = gaussian_filter(ocean_float, sigma=2)
   294→    c_grid = 0.6 - ocean_smooth * 1.8
   295→    c_grid = np.clip(c_grid, -1.2, 1.0)
   296→
   297→    # ─── Biome parameter table (matches BiomeParameters.java) ───
   298→    BIOME_PARAMS = [
   299→        # (name, T, H, C, E, W)
   300→        ('plains',            0.3,  0.0,  0.3,  0.5,  0.0),
   301→        ('sunflower_plains',  0.3,  0.0,  0.3,  0.5,  1.0),
   302→        ('forest',           0.4,  0.6,  0.3,  0.0,  0.0),
   303→        ('flower_forest',    0.4,  0.6,  0.3,  0.0,  1.0),
   304→        ('birch_forest',      0.6,  0.6,  0.3,  0.0,  0.0),
   305→        ('dark_forest',       0.7,  0.8,  0.3,  0.0,  0.0),
   306→        ('taiga',           -0.2,  0.3,  0.3,  0.0,  0.0),
   307→        ('snowy_taiga',      -0.3,  0.4,  0.3,  0.3,  0.0),
   308→        ('old_growth_spruce_taiga', -0.3, 0.8, 0.3, 0.3, 0.0),
   309→        ('old_growth_pine_taiga',   -0.3, 0.8, 0.3, 0.3, 0.0),
   310→        ('snowy_plains',     -0.5,  0.0,  0.3,  0.5,  0.0),
   311→        ('ice_spikes',       -0.5,  0.0,  0.3,  0.5,  1.0),
   312→        ('grove',           -0.2,  0.8,  0.3, -0.3,  0.0),
   313→        ('meadow',            0.5,  0.8,  0.3,  0.0,  0.0),
   314→        ('jagged_peaks',     -0.7,  0.9,  0.6, -1.0,  0.0),
   315→        ('frozen_peaks',     -0.7,  0.9,  0.6, -1.0,  0.0),
   316→        ('stony_peaks',       1.0,  0.3,  0.6, -1.0,  0.0),
   317→        ('snowy_slopes',     -0.3,  0.8,  0.3, -0.5,  0.0),
   318→        ('savanna',          0.8, -0.3,  0.3,  0.3,  0.0),
   319→        ('savanna_plateau',  0.8, -0.3,  0.6,  0.5,  0.0),
   320→        ('windswept_savanna', 1.0, -0.3,  0.3, -0.5,  0.0),
   321→        ('desert',            1.0, -0.5,  0.3,  0.8,  0.0),
   322→        ('badlands',          1.0, -0.5,  0.6, -0.5,  0.0),
   323→        ('wooded_badlands',    1.0, -0.3,  0.6, -0.5,  0.0),
   324→        ('jungle',            0.9,  0.9,  0.3,  0.0,  0.0),
   325→        ('sparse_jungle',      0.9,  0.9,  0.3,  0.0,  1.0),
   326→        ('bamboo_jungle',     0.9,  0.9,  0.3,  0.0,  0.5),
   327→        ('mangrove_swamp',    0.8,  0.9, -0.1,  0.0,  0.5),
   328→        ('windswept_hills',   0.2,  0.0,  0.3, -0.5,  0.0),
   329→        ('windswept_forest',  0.2,  0.3,  0.3, -0.5,  0.0),
   330→        ('swamp',             0.8,  0.9, -0.1,  0.0,  0.0),
   331→        ('cherry_grove',      0.5,  0.8,  0.3,  0.0,  1.0),
   332→    ]
   333→    # Pre-convert to numpy arrays for vectorized distance computation
   334→    bp_t  = np.array([p[1] for p in BIOME_PARAMS], dtype=np.float32)
   335→    bp_h  = np.array([p[2] for p in BIOME_PARAMS], dtype=np.float32)
   336→    bp_c  = np.array([p[3] for p in BIOME_PARAMS], dtype=np.float32)
   337→    bp_e  = np.array([p[4] for p in BIOME_PARAMS], dtype=np.float32)
   338→    bp_w  = np.array([p[5] for p in BIOME_PARAMS], dtype=np.float32)
   339→    bp_names = [p[0] for p in BIOME_PARAMS]
   340→
   341→    # Vectorized nearest-point matching for all land pixels
   342→    print("    Calculating biome parameters (vectorized)...")
   343→    land_y, land_x = np.where(land_mask)
   344→    if len(land_y) > 0:
   345→        t_vals = temp_norm_grid[land_y, land_x]  # (N,)
   346→        h_vals = hum_norm_grid[land_y, land_x]
   347→        c_vals = c_grid[land_y, land_x]
   348→        e_vals = ero_norm_grid[land_y, land_x]
   349→        # w = 0.0 for all
   350→        # Distance: (N, num_biomes)
   351→        dist = ((t_vals[:, None] - bp_t[None, :]) ** 2 +
   352→               (h_vals[:, None] - bp_h[None, :]) ** 2 +
   353→               (c_vals[:, None] - bp_c[None, :]) ** 2 +
   354→               (e_vals[:, None] - bp_e[None, :]) ** 2 +
   355→               bp_w[None, :] ** 2)
   356→        best_idx = np.argmin(dist, axis=1)
   357→        best_biomes = [bp_names[i] for i in best_idx]
   358→
   359→        for i in range(len(land_y)):
   360→            biome_name = best_biomes[i]
   361→            if biome_name in BIOME_COLORS:
   362→                biome_map[land_y[i], land_x[i]] = BIOME_COLORS[biome_name]
   363→            else:
   364→                biome_map[land_y[i], land_x[i]] = BIOME_COLORS['plains']
   365→
   366→    # Статистика
   367→    total = H * W
   368→    print(f"    Карта биомов: {W}x{H}")
   369→    print(f"    Статистика покрытия:")
   370→    print(f"      deep_ocean:  {deep_ocean_mask.sum():8d} ({100.0*deep_ocean_mask.sum()/total:.1f}%)")
   371→    print(f"      ocean:       {shallow_ocean_mask.sum():8d} ({100.0*shallow_ocean_mask.sum()/total:.1f}%)")

   387→def gray_to_height(gray):
   388→    """Преобразовать gray (0..255) в Y координату.
   389→
   390→    Кусочно-линейная формула (v3, совпадает с MapData.grayToHeight):
   391→      gray 0..72   → Y -30..63   (океан)
   392→      gray 72..255 → Y 63..200   (суша)
   393→    """
   394→    if gray <= SEA_LEVEL_GRAY:
   395→        t = gray / SEA_LEVEL_GRAY
   396→        return MIN_OCEAN + int(round(t * (SEA_LEVEL - MIN_OCEAN)))
   397→    else:
   398→        t = (gray - SEA_LEVEL_GRAY) / (255 - SEA_LEVEL_GRAY)
   399→        return SEA_LEVEL + int(round(t * (MAX_TERRAIN - SEA_LEVEL)))

   402→def get_pixel_info(maps, biome_map, px, pz):
   403→    """Получить подробную информацию о пикселе."""
   404→    info = {}
   405→
   406→    # Height (v3): gray 0 (Марианская) → Y -30, gray 72 → Y 63 (море), gray 255 → Y 200
   407→    gray = int(maps['height'][pz, px])
   408→    info['height_gray'] = gray
   409→    info['height_y'] = gray_to_height(gray)
   410→    # Legacy height-based ocean (for comparison only)
   411→    info['is_ocean_by_height'] = info['height_y'] < SEA_LEVEL
   412→    info['is_deep_ocean_by_height'] = info['height_y'] < DEEP_OCEAN_Y
   413→
   414→    # Humidity
   415→    info['humidity'] = int(maps['humidity'][pz, px])
   416→
   417→    # Temperature RGB → climate zone
   418→    rgb = maps['temperature'][pz, px]
   419→    info['temp_rgb'] = (int(rgb[0]), int(rgb[1]), int(rgb[2]))
   420→    climate, dist = detect_climate(rgb)
   421→    info['climate'] = climate
   422→    info['climate_temp'] = CLIMATE_TEMPS[climate]
   423→    info['climate_dist'] = round(dist, 2)
   424→
   425→    # Sea/ocean — v4: МАСКА (alpha>0 = океан)
   426→    sea_pixel = maps['sea_ocean'][pz, px]
   427→    info['sea_rgba'] = (int(sea_pixel[0]), int(sea_pixel[1]),
   428→                         int(sea_pixel[2]), int(sea_pixel[3]))
   429→    info['is_ocean'] = (sea_pixel[3] > 0) and (sea_pixel[0] < WATER_RED_THRESHOLD)
   430→    # Deep ocean: ocean AND height Y < DEEP_OCEAN_Y
   431→    deep_ocean_gray = int(round((DEEP_OCEAN_Y - MIN_OCEAN) * SEA_LEVEL_GRAY / (SEA_LEVEL - MIN_OCEAN)))
   432→    info['is_deep_ocean'] = info['is_ocean'] and (gray <= deep_ocean_gray)
   433→
   434→    # Lakes — alpha-aware (СОВПАДАЕТ с Java)
   435→    lake_pixel = maps['lakes'][pz, px]
   436→    info['lake_rgba'] = (int(lake_pixel[0]), int(lake_pixel[1]),
   437→                         int(lake_pixel[2]), int(lake_pixel[3]))
   438→    info['is_lake'] = (lake_pixel[3] > 0) and (lake_pixel[0] < WATER_RED_THRESHOLD)
   439→
   440→    # Rivers — alpha-aware (СОВПАДАЕТ с Java)
   441→    river_pixel = maps['rivers'][pz, px]
   442→    info['river_rgba'] = (int(river_pixel[0]), int(river_pixel[1]),
   443→                          int(river_pixel[2]), int(river_pixel[3]))
   444→    info['is_river'] = (river_pixel[3] > 0) and (river_pixel[0] < WATER_RED_THRESHOLD)
   445→
   446→    # Erosion
   447→    info['erosion'] = int(maps['erosion'][pz, px])
   448→
   449→    # Biome (determine from biome_map color)
   450→    biome_color = tuple(int(c) for c in biome_map[pz, px])
   451→    info['biome_color'] = biome_color
   452→    info['biome'] = None
   453→    for name, color in BIOME_COLORS.items():
   454→        if biome_color == color:
   455→            info['biome'] = name
   456→            break
   457→
   458→    # Определяем, нужно ли менять температуру из-за высоты (>80)
   459→    alt_temp = info['climate_temp']
   460→    if info['height_y'] > 80:
   461→        alt_temp -= (info['height_y'] - 80) * 0.001
   462→        alt_temp = max(0.0, alt_temp)
   463→    info['altitude_temp'] = round(alt_temp, 4)
   464→
   465→    return info

   580→def save_region(original_img, x, z, half_size):
   581→    """Сохранить область для последующего восстановления."""
   582→    key = (x, z)
   583→    if key not in saved_regions:
   584→        x1 = max(0, x - half_size)
   585→        z1 = max(0, z - half_size)
   586→        x2 = min(original_img.width, x + half_size + 1)
   587→        z2 = min(original_img.height, z + half_size + 1)
   588→        saved_regions[key] = original_img.crop((x1, z1, x2, z2))
   589→

   591→def restore_region(canvas, original_img, x, z, half_size):
   592→    """Восстановить область (затереть старый маркер)."""
   593→    key = (x, z)
   594→    if key in saved_regions:
   595→        region = saved_regions[key]
   596→        x1 = max(0, x - half_size)
   597→        z1 = max(0, z - half_size)
   598→        canvas.paste(region, (x1, z1))
   599→        del saved_regions[key]

   602→def draw_marker(draw, px, pz):
   603→    """Нарисовать маркер (крест + круг + буква P)."""
   604→    s = MARKER_SIZE
   605→    cs = CROSSHAIR_SIZE
   606→
   607→    # Crosshair
   608→    draw.line([(px - cs, pz), (px + cs, pz)], fill=MARKER_COLOR_OUTER, width=2)
   609→    draw.line([(px, pz - cs), (px, pz + cs)], fill=MARKER_COLOR_OUTER, width=2)
   610→
   611→    # Внешний круг
   612→    draw.ellipse([(px - s, pz - s), (px + s, pz + s)],
   613→                 outline=MARKER_COLOR_OUTER, width=2)
   614→
   615→    # Центральная точка
   616→    draw.ellipse([(px - 3, pz - 3), (px + 3, pz + 3)],
   617→                 fill=MARKER_COLOR_INNER, outline=MARKER_COLOR_OUTER, width=1)
   618→
   619→    # Буква P (Player)
   620→    try:
   621→        font = ImageFont.truetype("arial.ttf", 16)
   622→    except (OSError, IOError):
   623→        try:
   624→            font = ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", 14)
   625→        except (OSError, IOError):
   626→            font = ImageFont.load_default()
   627→
   628→    text = MARKER_TEXT
   629→    bbox = draw.textbbox((px, pz), text, font=font)
   630→    tw = bbox[2] - bbox[0]
   631→    th = bbox[3] - bbox[1]
   632→    draw.text((px - tw // 2, pz - s - th - 4), text, fill=MARKER_COLOR_OUTER, font=font)
